pipeline {
    agent any
    tools {
        maven 'Maven3'   // Must match name in Global Tool Config
    }
    stages {
        stage('Checkout') {
            steps {
                git branch: 'staging', url: 'https://github.com/suhas19-Git/Devops-task.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean install'
            }
        }
        stage('Test') {
            steps {
                sh 'mvn test'
            }
        }
    }
}
