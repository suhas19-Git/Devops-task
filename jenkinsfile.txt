pipeline {
    agent any

    stages {

        stage('IP Address') {
            steps {
                sh '''
                echo "===== IP ADDRESS ====="
                hostname -I
                '''
            }
        }

        stage('Disk Usage') {
            steps {
                sh '''
                echo "===== DISK USAGE ====="
                df -h
                '''
            }
        }

        stage('CPU Details') {
            steps {
                sh '''
                echo "===== CPU DETAILS ====="
                lscpu
                '''
            }
        }

        stage('Memory Usage') {
            steps {
                sh '''
                echo "===== MEMORY USAGE ====="
                free -h
                '''
            }
        }
    }

    post {
        always {
            echo "System info pipeline completed ✅"
        }
    }
}
